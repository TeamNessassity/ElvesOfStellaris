trait_elven_lhuren_celestial_grace = {
	force_councilor_trait = yes
	inline_script = {
		script = trait/icon_specialist
		CLASS = leader
		ICON = "GFX_leader_trait_elven_lhuren_celestial_grace"
		RARITY = veteran
		COUNCIL = yes
		TIER = none
	}
	custom_subtitle = trait_elven_lhuren_celestial_grace_subtitle

	# Councilors without a valid home planet (Unlikely to happen, but possible in case of script-spawned leaders, or ones whose worlds got zonked)
	triggered_councilor_modifier = {
		potential = {
			inline_script = {
				script = traits/eos_lhuren_trigger_leader_planet_invalid
			}
		}
		planet_stability_mult = -0.1
	}

	# Councilors with home planets in other empires give envoys and diplo weight
	triggered_councilor_modifier = {
		potential = {
			inline_script = {
				script = traits/eos_lhuren_trigger_leader_planet_foreign
			}
		}
		envoys_add = 1
	}
	triggered_councilor_modifier = {
		potential = {
			leader_class = commander
			inline_script = {
				script = traits/eos_lhuren_trigger_leader_planet_foreign
			}
		}
		diplo_weight_naval_mult = 0.01
		mult = root.trigger:has_total_skill
	}
	triggered_councilor_modifier = {
		potential = {
			leader_class = scientist
			inline_script = {
				script = traits/eos_lhuren_trigger_leader_planet_foreign
			}
		}
		diplo_weight_technology_mult = 0.01
		mult = root.trigger:has_total_skill
	}
	triggered_councilor_modifier = {
		potential = {
			leader_class = official
			inline_script = {
				script = traits/eos_lhuren_trigger_leader_planet_foreign
			}
		}
		diplo_weight_economy_mult = 0.01
		mult = root.trigger:has_total_skill
	}

	# Councilors give Celestial Offices to their home planets, 100 per seat
	# Their Effective Leader Skill (important distinction: it is NOT the same as Effective Councilor Skill) additionally increases Celestials' Job Efficiency

	# Commanders host Celestial Strategists
	triggered_background_planet_modifier = {
		potential = {
			leader_class = commander
			inline_script = {
				script = traits/eos_lhuren_trigger_leader_planet_owned
			}
		}
		job_elven_dragon_strategist_add = 100
	}
	triggered_background_planet_modifier = {
		potential = {
			leader_class = commander
			inline_script = {
				script = traits/eos_lhuren_trigger_leader_planet_owned
			}
		}
		pop_elven_dragon_strategist_bonus_workforce_mult = 0.1
		mult = root.trigger:has_total_skill
	}

	# Scientists host Celestial Scholars
	triggered_background_planet_modifier = {
		potential = {
			leader_class = scientist
			inline_script = {
				script = traits/eos_lhuren_trigger_leader_planet_owned
			}
		}
		job_elven_dragon_researcher_add = 100
	}
	triggered_background_planet_modifier = {
		potential = {
			leader_class = scientist
			inline_script = {
				script = traits/eos_lhuren_trigger_leader_planet_owned
			}
		}
		pop_elven_dragon_researcher_bonus_workforce_mult = 0.1
		mult = root.trigger:has_total_skill
	}

	# Officials host Celestial Philosophers
	triggered_background_planet_modifier = {
		potential = {
			leader_class = official
			inline_script = {
				script = traits/eos_lhuren_trigger_leader_planet_owned
			}
		}
		job_elven_dragon_priest_add = 100
	}
	triggered_background_planet_modifier = {
		potential = {
			leader_class = official
			inline_script = {
				script = traits/eos_lhuren_trigger_leader_planet_owned
			}
		}
		pop_elven_dragon_priest_bonus_workforce_mult = 0.1
		mult = root.trigger:has_total_skill
	}

	# Ruler gives flat +50% efficiency on top of scaling bonuses
	triggered_background_planet_modifier = {
		potential = {
			is_ruler = yes
			leader_class = commander
			inline_script = {
				script = traits/eos_lhuren_trigger_leader_planet_owned
			}
		}
		pop_elven_dragon_strategist_bonus_workforce_mult = 0.5
	}
	triggered_background_planet_modifier = {
		potential = {
			is_ruler = yes
			leader_class = scientist
			inline_script = {
				script = traits/eos_lhuren_trigger_leader_planet_owned
			}
		}
		pop_elven_dragon_researcher_bonus_workforce_mult = 0.5
	}
	triggered_background_planet_modifier = {
		potential = {
			is_ruler = yes
			leader_class = official
			inline_script = {
				script = traits/eos_lhuren_trigger_leader_planet_owned
			}
		}
		pop_elven_dragon_priest_bonus_workforce_mult = 0.5
