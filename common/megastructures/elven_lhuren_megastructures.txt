			is_inside_border = from
		}
		custom_tooltip = {
			fail_text = "requires_colonized_planet_orbital_ring"

			fail_text = "elven_lhuren_eternal_array_requires_immortality_fail"

			NOT = { has_global_flag = eos_lhuren_immortality_lost }
		}
	}

	on_build_start = {}
	on_build_cancel = {}

	on_build_complete = {
		fromfrom = { save_global_event_target_as = lhuren_eternal_array }

		from = {
			country_event = { id = elven_lhuren_origin_events.1051 }
		}
	}
}
