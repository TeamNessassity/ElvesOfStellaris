d_drow_planetary_zhevalshath = {
	is_for_colonizable = yes
	icon = d_drow_planetary_zhevalshath

	category = deposit_cat_rare

	planet_modifier = { 
		planet_housing_add = 1000
		planet_amenities_add = 1000
	}
	
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				has_active_tradition = tr_domination_imperious_architecture
			}					
		}
		modifier = {
			planet_housing_add = 500
			planet_amenities_add = 500
		}
	}
	
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { is_gestalt = yes }
		}
		job_patrol_drone_add = @building_static_jobs
		job_warrior_drone_add = @building_static_jobs_low
	}
	
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { 
				NOR = {
					has_authority = auth_corporate
					has_authority = auth_imperial
					has_civic = civic_aristocratic_elite
				}
			}					
		}
		modifier = {
			job_politician_add = @building_static_jobs_low
			job_enforcer_add = @building_static_jobs_low
			job_soldier_add = @building_static_jobs_low
		}
	}
	
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { has_authority = auth_corporate }
		}
		modifier = {
			job_executive_add = @building_static_jobs_low
			job_enforcer_add = @building_static_jobs_low
			job_soldier_add = @building_static_jobs_low
		}
	}	
	
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { 
				OR = {
					has_civic = civic_aristocratic_elite
					has_authority = auth_imperial
				}
			}
		}
		modifier = {
			job_noble_add = @building_static_jobs_low
			job_enforcer_add = @building_static_jobs_low
			job_soldier_add = @building_static_jobs_low
		} 
	}

	drop_weight = {
		weight = 0
	}
}
