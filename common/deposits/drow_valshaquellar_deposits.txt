d_drow_collapsed_spire = {
	time = 480
	icon = d_drow_broken_spire
	
	is_for_colonizable = yes
	category = deposit_cat_rare_blocker
	
	potential = { always = no }
	
	planet_modifier = { 
		planet_max_districts_add = -1
	}

	use_weights_for_blocker_swap_types = no

	on_cleared = {
		custom_tooltip = add_zone_central_spire_tt
		hidden_effect = {
			add_zone = {
				district = 0
				zone = zone_central_spire
				zone_slot = 1
				replace = yes
			}
		}
	}

	resources = {
		category = deposit_blockers
		cost = {
			energy = 2000
		}
	}
}

d_drow_central_spire = {
	is_for_colonizable = yes
	icon = d_drow_central_spire
	category = deposit_cat_rare

	potential = { always = no }
	
	planet_modifier = { 
		planet_jobs_society_research_produces_mult = 0.15
		planet_jobs_engineering_research_produces_mult = 0.15
		planet_jobs_physics_research_produces_mult = 0.15
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { is_regular_empire = yes }
		}
		modifier = {
			job_physicist_add = 800
			job_biologist_add = 800
			job_engineer_add = 800
		}
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { is_hive_empire = yes }
		}
		modifier = {
			job_calculator_physicist_add = 800
			job_calculator_biologist_add = 800
			job_calculator_engineer_add = 800
		}
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { is_machine_empire = yes }
		}
		modifier = {
			job_calculator_physicist_add = 800
			job_calculator_biologist_add = 800
			job_calculator_engineer_add = 800
		}
	}
}
