## ARCANE SOCIETIES ##
civic_elven_arcane_minded = {
	icon = "gfx/interface/icons/elven/civic_elven_arcane.dds"
	potential = { 
		ethics = { NOT = { value = ethic_gestalt_consciousness } }
		authority = { NOT = { value = auth_corporate } }
	}	
	
	possible = {
		ethics = {
			OR = {
				text = civic_tooltip_spiritualist
				value = ethic_spiritualist
				value = ethic_fanatic_spiritualist
			}
		}
	}
	
	pickable_at_start = yes
	
	random_weight = {
		base = 0
	}
	
	modifier = {
		ship_anomaly_generation_chance_mult = 0.1
	}
	
	description = "civic_tooltip_arcane_minded_effects"
}
