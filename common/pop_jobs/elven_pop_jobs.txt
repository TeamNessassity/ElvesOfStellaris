elven_dragon_researcher = {
	category = ruler

	possible_pre_triggers = {
		has_owner = yes
		is_enslaved = no
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible_precalc = can_fill_ruler_job

	tags = { unity research }

	resources = {
		category = planet_eos_celestials
		produces = {
			unity = 1
			physics_research = 1
			mult = trigger:capital_tier
		}
		produces = {
			unity = 1
			physics_research = 1
			mult = trigger:has_planetary_ascension_tier
		}
		upkeep = {
			trade = 2
			mult = trigger:capital_tier
		}
	}

	triggered_planet_modifier = {
		pop_biologist_bonus_workforce_mult = 0.1
		pop_physicist_bonus_workforce_mult = 0.1
		pop_engineer_bonus_workforce_mult = 0.1
		pop_healthcare_bonus_workforce_mult = 0.1
		pop_roboticist_bonus_workforce_mult = 0.1
	}

	weight = {
		weight = @ruler_job_weight
		modifier = {
			factor = 2
			has_trait = trait_elven_lhuren_immortality
		}
		modifier = {
			factor = 0.1
			has_citizenship_rights = no
			NOT = { has_trait = trait_mechanical }
		}
		# crisis purge
		modifier = {
			factor = 0.01
			exists = planet
			exists = planet.controller
			planet.controller = {
				OR = {
					is_country_type = swarm
					is_country_type = ai_empire
				}
			}
		}
	}

	inline_script = "jobs/automodding_priority_research"
}
