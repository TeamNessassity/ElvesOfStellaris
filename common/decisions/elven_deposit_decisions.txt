decision_elven_remove_sentient_dragons = {
	owned_planets_only = yes
	sound = event_administrative_work
	icon = elven_remove_sentient_dragons
	enactment_time = 720

	resources = {
		category = decisions
		cost = {
			energy = 2000
		}
	}

	potential = {
		has_deposit = d_elven_planetary_sentient_dragons
		owner = {
			OR = {
				is_gestalt = yes 
				has_valid_civic = civic_fanatic_purifiers
			}
		}
	}

	allow = {
		owner = { has_technology = tech_dangerous_wildlife }
	}

	effect = {
		remove_deposit = d_elven_planetary_sentient_dragons

		owner = {
			if = {
				limit = { 
					OR = {
						has_valid_civic = civic_fanatic_purifiers
						has_valid_civic = civic_machine_terminator
					}
				}

				add_resource = { unity = 1000 }
			}

			if = {
				limit = { has_valid_civic = civic_hive_devouring_swarm }

				add_resource = { 
					food = 500
					society_research = 500
				}
			}
		}

		set_carrier_flag = removed_sentient_dragons
		eos_lhuren_immortality_remove = yes
	}

	ai_weight = {
		weight = 1000
	}
}
