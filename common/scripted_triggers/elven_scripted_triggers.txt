# Xenocidal

drow_events_is_xenocidal = {
	is_species_class = DROW
	pop_has_trait = trait_drow_xenocidal
	NOT = {
		is_enslaved = yes
		is_being_purged = yes
	}
}

drow_events_is_xenocidable = {
	is_enslaved = no
	is_robot_pop = no
	is_being_purged = no
	NOT = {
		pop_has_trait = trait_drow_xenocidal
		pop_has_trait = trait_presapient_proles		
	}
}

drow_events_has_xenocidable_ethic = {
	OR = {
		pop_has_ethic = ethic_xenophile
		pop_has_ethic = ethic_pacifist
		pop_has_ethic = ethic_egalitarian
	}
}
