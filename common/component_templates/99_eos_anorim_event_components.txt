	ai_tags = { debuffer_role debuffer_role_stealth }

	hidden = yes
	hide_damage_values_from_tooltip = yes

	power = 0
	damage = { min = 0 max = 0 }
	windup = { min = 10 max = 10 }
	total_fire_time = 10
	range = 30
	accuracy = 1.0
	tracking = 0.5

	target_type = target_enemies
	target_focus = spread
	on_hit = {
		hidden_effect = {
			if = {
				limit = {
					exists = this
					is_civilian = no
					NOT = {
						is_ship_class = shipclass_starbase
					}
				}
				random = {
					chance = value:weaver_confuser_chance|TIER|5|
					set_confused = yes
					set_ship_flag = is_confused
					ship_event = {
						id = bioship.1
						days = 5
						random = 5
					}
				}
			}
		}
		custom_tooltip = "bio_confuser_$TIER$_tt"
	}
}
