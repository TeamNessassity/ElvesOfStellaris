generate_elven_primitives_on_planet = {
	optimize_memory
	select_pre_ftl_origin = yes
	create_elven_pre_ftl_species = yes
	create_elven_pre_ftl_empire = yes
	setup_pre_ftl_planet = yes
	pre_ftl_clean_up_effect = yes
}

create_elven_pre_ftl_species = {

	if = {
		limit = {
			has_planet_flag = pre_ftl_ocean_paradise
		}
		create_species = {
			name = random
			class = ELVEN
			portrait = elven
			homeworld = THIS
			traits = {
				trait = trait_aquatic
				trait = trait_elven_true
				trait = random_traits
			}
		}
	}
	else_if = {
		limit = {
			has_planet_flag = pre_ftl_subterranean
		}
		create_species = {
			name = random
			class = ELVEN
			portrait = elven
			homeworld = THIS
			traits = {
				trait = trait_cave_dweller
				trait = trait_elven_true
				trait = random_traits
			}
		}
	}
	else_if = {
		limit = {
			has_planet_flag = pre_ftl_void_dwellers
		}
		create_species = {
			name = random
			class = ELVEN
			portrait = elven
			homeworld = THIS
			traits = {
				trait = trait_void_dweller_1
				trait = trait_elven_true
				trait = random_traits
			}
		}
	}
	else = {
		create_species = {
			name = random
			class = ELVEN
			portrait = elven
			homeworld = THIS
			traits = {
				trait = trait_elven_true
				trait = random_traits
			}
		}
	}
}

create_elven_pre_ftl_empire = {
	random_list = {
		# Industrial Civilization
		35 = {
			create_late_elven_primitive_empire = yes
		}
		# Medieval Civilization
		65 = {
			create_early_elven_primitive_empire = yes
		}
	}
}

		}
		species = last_created_species
		ethos = random

			icon = {
				category = "pre_ftl"
				file = "preftl_stone_age.dds"
