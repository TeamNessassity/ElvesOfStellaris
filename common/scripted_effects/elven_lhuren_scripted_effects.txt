# Add/Remove the Celestial Favour Trait from leader if eligible
# this = leader
elven_verify_celestial_favor_trait = {
	if = {
		limit = {
			owner = {
				has_civic = civic_elven_celestial_grace
			}
			OR = {
				is_ruler = yes
				is_councilor = yes
			}
		}
		add_trait = {
			trait = trait_elven_lhuren_celestial_grace
			show_message = no
		}
	}
	else = {
		remove_trait = trait_elven_lhuren_celestial_grace
	}
}

# Eternal Array shield management

# this = planet
eos_lhuren_shield_state = {
	set_variable = {
		which = eos_lhu_var_shield_state
		value = $VALUE$
	}

	# Lhumar shield
	if = {
		limit = {
			check_variable = {
				which = eos_lhu_var_shield_state
				value = 1
			}
		}
		eos_lhuren_shield_set_ambient_entity = { TYPE = "lhu_array_eternity_shield_ambient_broken" }
	}
	else_if = {
		limit = {
			check_variable = {
				which = eos_lhu_var_shield_state
				value = 2
			}
		}
		eos_lhuren_shield_set_ambient_entity = { TYPE = "lhu_array_eternity_shield_ambient_partial" }
	}
	else_if = {
		limit = {
			check_variable = {
				which = eos_lhu_var_shield_state
				value = 3
			}
		}
		eos_lhuren_shield_set_ambient_entity = { TYPE = "lhu_array_eternity_shield_ambient_full" }
	}

	# Generic shield
	else_if = {
		limit = {
			check_variable = {
				which = eos_lhu_var_shield_state
				value = -1
			}
		}
		eos_lhuren_shield_set_ambient_entity = { TYPE = "lhu_array_eternity_shield_ambient_weak" }
	}

	# Remove the shield completely
	else_if = {
		limit = {
			check_variable = {
				which = eos_lhu_var_shield_state
				value = 0
			}
		}
		eos_lhuren_shield_clear_ambient_entity = yes
	}
}

# this = planet
eos_lhuren_shield_set_ambient_entity = {
	solar_system = {
		every_system_ambient_object = {
			limit = {
				has_ambient_object_flag = lhu_shield_of_@prevprev
			}
			destroy_ambient_object = this
		}
	}

	create_ambient_object = {
		type = $TYPE$
		location = this
		# I just learned this thing exists and my life is changed. NO MORE PLANET OFFSETS!
		use_3d_location = yes
		entity_scale_to_size = yes
	}
	last_created_ambient_object = {
		set_ambient_object_flag = lhu_shield_of_@prev
	}
}

# this = planet
eos_lhuren_shield_clear_ambient_entity = {
	solar_system = {
		every_system_ambient_object = {
			limit = {
				has_ambient_object_flag = lhu_shield_of_@prevprev
			}
			destroy_ambient_object = this
		}
	}
}

# this = planet
eos_lhuren_shield_state_modifier = {
	remove_modifier = elven_eternal_array_modifier
	remove_modifier = elven_eternal_array_broken_modifier
	remove_modifier = elven_eternal_array_encased_modifier
	remove_modifier = elven_eternal_array_repurposed_modifier
	add_modifier = {
		modifier = $MODIFIER$
		days = -1
	}
}

# Lhuren: Remove Immortality

# this = planet
# Since this can trigger off planet destruction, we cannot assume ownership by the time it occurs.
eos_lhuren_immortality_remove = {
	if = {
		limit = {
			NOT = { has_global_flag = eos_lhuren_immortality_lost }
			NOT = { has_global_flag = elven_lhuren_pact_enhanced }
		}
		set_global_flag = eos_lhuren_immortality_lost

		# Kill a few pops to make it a bit more noticeable
		every_country = {
			every_owned_pop_group = {
				limit = {
					has_trait = trait_elven_lhuren_immortality
				}
				# Perfectly balanced, as all things should be
				kill_pop_group = {
					pop_group = this
					percentage = 0.5
				}
			}
		}

		every_galaxy_planet = {
			limit = {
				has_deposit = d_elven_planetary_sentient_dragons
			}
			remove_deposit = d_elven_planetary_sentient_dragons
		}

		every_galaxy_species = {
			limit = {
				has_trait = trait_elven_lhuren_immortality
			}
			change_species_characteristics = {
				remove_trait = trait_elven_lhuren_immortality
			}
		}

		every_country = {
			limit = {
				has_origin = origin_elven_celestial_throne
			}
			end_event_chain = elven_lhuren_eternal_array_chain
			remove_relic = r_elven_flaming_pearl
			country_event = {
				id = elven_lhuren_origin_events.301
			}
		}
	}
}
