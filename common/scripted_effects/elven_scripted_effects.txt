eos_create_treants = {
	optimize_memory

	create_species = {
		name = "Treant"
		plural = "Treants"
		class = PLANT
		portrait = pla14
		namelist = PLANT4
		homeworld = THIS

		traits = {
			trait = "trait_syncretic_proles"
			trait = "trait_very_strong"
			trait = "trait_resilient"
			trait = "trait_slow_breeders"
			trait = "trait_sedentary"
		}

		effect = {
			set_species_flag = elven_arcadian_symbiosis_treants

			save_global_event_target_as = elven_treant_species
		}
	}

	random_owned_pop_group = {
		limit = {
			pop_group_size >= 800
			is_pop_category = worker
		}
		kill_pop_group = {
			pop_group = this
			amount = 800
		}
	}

	while = {
		count = 8
		create_pop_group = {
			species = event_target:elven_treant_species
			category = worker
		}
	}

	recalculate_treant_habitability_bonus = yes
}

recalculate_treant_habitability_bonus = {
	remove_modifier = elven_treant_harmony

	set_variable = { which = elven_treant_count value = 0 }
	export_trigger_value_to_variable = {
		trigger = count_owned_pop_amount
		variable = elven_treant_count
		parameters = {
			limit = { species = { has_species_flag = elven_arcadian_symbiosis_treants } }
		}
		rounded = yes
	}
	divide_variable = {
		which = elven_treant_count
		value = 100
	}

	add_modifier = {
		modifier = elven_treant_harmony
		days = -1
		multiplier = elven_treant_count
	}
}

recalculate_unity_to_research_conversion = {
	remove_modifier = elven_unity_to_research_conversion
	remove_modifier = elven_unity_to_research_conversion_research

	# Export total unity income (with country-wide % multipliers) into a variable, throwing away the unity penalty from the previous month calculation.
	# The penalty must be discarded to avoid a feedback loop.
	set_variable = { which = elven_unity_production value = 0 }
	export_resource_income_to_variable = {
		resource = unity
		variable = elven_unity_production
	}
	change_variable = { 
		which = elven_unity_production
		value = elven_current_unity_penalty
	}

	# Take only 20% of unity income, as specified in the civic itself.
	divide_variable = { 
		which = elven_unity_production
		value = 5
	}

	# Create a flat research country modifier based on total unity income. 
	add_modifier = {
		modifier = elven_unity_to_research_conversion_research
		days = -1
		multiplier = elven_unity_production
	}

	# Read country-wide unity % multipliers into another variable.
	export_modifier_to_variable = {
		modifier = country_unity_produces_mult
		variable = elven_unity_production_mult
	}
	change_variable = { 
		which = elven_unity_production_mult
		value = 1
	}

	# Calculate BASE unity penalty by removing country-wide unity % multipliers.
	# If we don't do this, the game will apply these multipliers AGAIN, creating a potential feedback loop for subsequent calculations.
	divide_variable = { 
		which = elven_unity_production
		value = elven_unity_production_mult
	}
	
	# Store the unity penalty for the next month calculations.
	set_variable = { which = elven_current_unity_penalty value = elven_unity_production }

	# Apply the unity penalty as a modifier.
	# The game engine will apply country-wide unity % multipliers to the unity value from this modifier.
	add_modifier = {
		modifier = elven_unity_to_research_conversion
		days = -1
		multiplier = elven_unity_production
	}
}

# Galtaereth kills a percentage of the cursed pops and mocks you
# this = colony
# MESSAGE = message loc
# MESSAGE_DESC = message description loc
# PERCENTAGE = float (0.0 to 1.0)
eos_evergloom_tithe = {
	create_message = {
		type = $MESSAGE$
		localization = $MESSAGE_DESC$
		days = 30
		target = this.solar_system
		variable = {
			type = name
			localization = PLANET
			scope = this
		}
	}
	every_owned_pop_group = {
		limit = { has_trait = trait_drow_anorim_curse }
		kill_pop_group = {
			pop_group = this
			percentage = $PERCENTAGE$
		}
	}
}
