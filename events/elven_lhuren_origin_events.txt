# Ending B Mechanics: If the Orbital Ring is somehow destroyed/dismantled, break the shield and downgrade the Array so it can be repaired again.

event = {
	id = elven_lhuren_origin_events.511
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		exists = event_target:lhuren_eternal_array
		event_target:lhuren_eternal_array = {
			is_megastructure_type = elven_lhuren_eternal_array_restored
		}
		exists = event_target:lhuren_lhumar
		NOT = { exists = event_target:lhuren_eternal_array_starbase }
	}

	immediate = {
		event_target:lhuren_lhumar = {
			solar_system = {
				# Vanilla forcibly spawns a ruined ring if the system is abandoned while an orbital ring exists
				# One scenario where it can happen is if player embarks from Lhumar with their whole empire
				# Anyways, UGLY! let's get rid of it!
				every_megastructure = {
					limit = {
						is_megastructure_type = orbital_ring_ruined
					}
					delete_megastructure = this
				}

				# Now, switch the Array back to its ruined state
				# After Ending B, Lhuren can repair it anytime
				delete_megastructure = event_target:lhuren_eternal_array
				spawn_megastructure = {
					type = elven_lhuren_eternal_array_ruined
					planet = event_target:lhuren_lhumar
					owner = event_target:lhuren_lhumar.owner
					graphical_culture = elven_02

					init_effect = {
						save_global_event_target_as = lhuren_eternal_array
					}
				}
			}
		}
	}
}
