# Story: Lhuren used to be interstellar nomads who found the Dragon Cluster and the Dirac Sea Serpents that could manifest only within the nebula. They made a pact with them, gaining immortality and prosperity for as long as they protected Lhumar (a world quantum-entangled across dimensions with the serpents' homeworld).
# Objectives: Eternal Array malfunctions, game begins. Excavate two sites in the system for clues. Then, decide whether to remove the faulty shield module and repurpose the array, stabilise the module with limited resources or fully repair.

# Stage 0 (Gatekeeper Event)
# Start the event OR show an error message if cannot
country_event = {
	id = elven_lhuren_origin_events.100
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_origin = origin_elven_celestial_throne
	}
	
	immediate = {
		if = {
			limit = {
				NOT = { is_same_value = event_target:lhuren_capital_owner_of_@event_target:lhuren_lhumar }
			}
			country_event = {
				id = elven_lhuren_origin_events.0001
			}
		}
		else = {
			country_event = {
				id = elven_lhuren_origin_events.101
			}
		}
	}
}

		hidden_effect = {
			create_variable_if_not_exists = {
				VARIABLE = eos_lhu_var_mega_count
				VALUE_IF_CREATED = 0
			}
			change_variable = {
				which = eos_lhu_var_mega_count
				value = 5
			}
		}
	}
	
	option = {
		allow = {
			has_civic = civic_inwards_perfection
		}
		name = elven_lhuren_origin_events.1051.a
		if = {
			limit = { has_civic = civic_inwards_perfection }
			force_remove_civic = civic_inwards_perfection
			force_add_civic = civic_efficient_bureaucracy
		}
		ai_chance = {
			factor = 75
		}
	}
	
	option = {
		name = elven_lhuren_origin_events.1051.b
		ai_chance = {
			factor = 25
		}
		default_hide_option = yes
	}
}

# Ending C (Letting All of My Feelings Disappear)
# Remove Lhumar, Eternal Array, Flaming Pearl and Ruler. Offer to remove Inwards Perfection. Enhance the Timeless Trait.
country_event = {
	id = elven_lhuren_origin_events.1071
	title = "elven_lhuren_origin_events.1071.name"
	desc = "elven_lhuren_origin_events.1071.desc"
	picture = GFX_evt_eos_lhuren_lhumar
	show_sound = event_mystic_reveal

	is_triggered_only = yes
	
	immediate = {
	}

	after = {
		end_event_chain = elven_lhuren_eternal_array_chain
		set_country_flag = elven_lhuren_origin_ending_C
		add_modifier = {
			modifier = elven_lhuren_origin_ending_C_modifier
		}

		set_global_flag = elven_lhuren_pact_enhanced

		event_target:lhuren_lhumar = {
			hidden_effect = {
				# The initial Nirvana counter is based on the number of pops present on planet when shield goes up
				# It ticks up by 1 every month after that
				export_trigger_value_to_variable = {
					trigger = count_owned_pop_amount
					parameters = {
						limit = {
							has_trait = trait_elven_lhuren_immortality
						}
					}
					variable = eos_lhu_var_nirvana
				}
			}

			hidden_effect = {
				if = {
					limit = { exists = orbital_defence }
					delete_fleet = orbital_defence
				}
			}

			change_pc = pc_shielded
			set_planet_entity = {
				entity = elven_ocean_2_lhumar
			}
			set_planet_flag = lhuren_capital_encased
			clear_planet_modifiers = yes

			remove_modifier = elven_eternal_array_broken_modifier
			add_modifier = {
				modifier = elven_eternal_array_encased_modifier
			}

			hidden_effect = {
				eos_lhuren_shield_state = { VALUE = 3 }
			}
		}
		ruler = {
			kill_leader = {
				show_notification = no
			}
		}
		remove_relic = r_elven_flaming_pearl
	}
	
	option = {
		allow = {
			has_civic = civic_inwards_perfection
		}
		name = elven_lhuren_origin_events.1071.a
		if = {
			limit = { has_civic = civic_inwards_perfection }
			force_remove_civic = civic_inwards_perfection
			force_add_civic = civic_efficient_bureaucracy
		}
		ai_chance = {
			factor = 80
		}
	}
	
	option = {
		name = elven_lhuren_origin_events.1071.b
		ai_chance = {
			factor = 20
		}
		default_hide_option = yes
	}
}

# Planet glassed/changed to something uninhabitable other than shielded
planet_event = {
	id = elven_lhuren_origin_events.201
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		exists = event_target:lhuren_lhumar
		is_same_value = event_target:lhuren_lhumar
		OR = {
			is_planet_class = pc_nuked
			is_planet_class = pc_machine
			AND = {
				NOT = { is_planet_class = pc_shielded }
				habitable_planet = no
			}
		}
	}

	immediate = {
		eos_lhuren_immortality_remove = yes
	}
}

# Ending A Perk: When building or upgrading Megastructures, give a random planet a boost for a year

country_event = {
	id = elven_lhuren_origin_events.501
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		# Since it's now tied to a megastructure, which can change hands, the variable should be tracked by all countries just in case
	}

	immediate = {
		create_variable_if_not_exists = {
			VARIABLE = eos_lhu_var_mega_count
			VALUE_IF_CREATED = 0
		}
		change_variable = {
			which = eos_lhu_var_mega_count
			value = 1
		}

		if = {
			limit = { has_country_flag = elven_lhuren_origin_ending_A }
			random_owned_planet = {
				limit = {
					is_colony = yes
					NOT = { has_modifier = elven_lhuren_planet_boost }
				}
				add_modifier = {
					modifier = elven_lhuren_planet_boost
					years = 1
				}
			}
		}
	}
}
