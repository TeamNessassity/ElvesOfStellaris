namespace = elven_celestial_grace_events

# Give Celestial Grace when assigned to Empire Council
# Scope: Leader (after assignment)
leader_event = {
	id = elven_celestial_grace_events.201
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		OR = {
			owner = { has_civic = civic_elven_celestial_grace }
			has_trait = trait_elven_lhuren_celestial_grace
		}
	}

	immediate = {
		elven_verify_celestial_favor_trait = yes
	}
}

# Remove Celestial Grace when unassigned from Empire Council
# Scope: Leader (just before unassignment)
leader_event = {
	id = elven_celestial_grace_events.202
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		OR = {
			owner = { has_civic = civic_elven_celestial_grace }
			has_trait = trait_elven_lhuren_celestial_grace
		}
	}

	immediate = {
		# Since the leader is still technically on the council by the time the event fires, removal needs to be explicit
		remove_trait = trait_elven_lhuren_celestial_grace
	}
}

# Celestial Grace Maintenance
country_event = {
	id = elven_celestial_grace_events.203
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		OR = {
			has_civic = civic_elven_celestial_grace
			any_owned_leader = {
				has_trait = trait_elven_lhuren_celestial_grace
			}
		}
	}

	immediate = {
		every_owned_leader = {
			elven_verify_celestial_favor_trait = yes
		}
	}
}

# Celestial Grace Decisions/Edicts
# Rulers can abdicate or move Celestial Offices to the current Capital
country_event = {
	id = elven_celestial_grace_events.301
	title = "elven_celestial_grace_events.301.name"
	desc = "elven_celestial_grace_events.301.desc"
	picture = GFX_evt_eos_lhuren_lhumar
	show_sound = event_administrative_work
	is_triggered_only = yes

	trigger = {
		exists = ruler
		exists = capital_scope
	}

	immediate = {
	}

	after = {
	}
	
	option = {
		allow = {
			NOT = { exists = ruler.background_colony }
			resource_stockpile_compare = {
				resource = influence
				value >= 400
			}
		}
		name = elven_celestial_grace_events.301.a
		ruler = {
			save_event_target_as = lhu_current_ruler
		}
		event_target:lhu_current_ruler = {
			add_timed_trait = {
				trait = trait_elven_lhuren_concessions
				years = 50
			}
		}
		add_resource = {
			influence = -400
		}
		ai_chance = {
			factor = 40
		}
	}
	
	option = {
		allow = {
			exists = heir
			heir = {
				NOT = { has_trait = trait_elven_lhuren_taishanghuang }
			}
			resource_stockpile_compare = {
				resource = influence
				value >= 200
			}
		}
		name = elven_celestial_grace_events.301.b
		ruler = {
			save_event_target_as = lhu_current_ruler
		}
		if = {
			limit = { exists = heir }
			heir = {
				save_event_target_as = lhu_current_heir
			}
		}
		set_leader = event_target:lhu_current_heir
		event_target:lhu_current_ruler = {
			add_trait = {
				trait = trait_elven_lhuren_taishanghuang
			}
		}
		add_resource = {
			influence = -200
		}
		ai_chance = {
			factor = 40
		}
	}
	
	option = {
		allow = {
			any_owned_leader = {
				eos_lhu_valid_new_dynasty_ruler = yes
			}
			resource_stockpile_compare = {
				resource = influence
				value >= 200
			}
		}
		name = elven_celestial_grace_events.301.c
		ruler = {
			save_event_target_as = lhu_current_ruler
		}
		if = {
			limit = {
				any_owned_leader = {
					is_councilor = yes
					eos_lhu_valid_new_dynasty_ruler = yes
				}
			}
			random_owned_leader = {
				limit = {
					is_councilor = yes
					eos_lhu_valid_new_dynasty_ruler = yes
				}
				save_event_target_as = lhu_candidate_ruler
			}
		}
		else_if = {
			limit = {
				any_owned_leader = {
					is_councilor = no
					eos_lhu_valid_new_dynasty_ruler = yes
				}
			}
			random_owned_leader = {
				limit = {
					is_councilor = no
					eos_lhu_valid_new_dynasty_ruler = yes
				}
				save_event_target_as = lhu_candidate_ruler
			}
		}
		# Weird but there's no other way to just clear the heir slot
		set_heir = event_target:lhu_candidate_ruler
		set_leader = event_target:lhu_candidate_ruler
		event_target:lhu_current_ruler = {
			add_trait = {
				trait = trait_elven_lhuren_taishanghuang
			}
		}
		add_resource = {
			influence = -200
		}
		ai_chance = {
			factor = 20
		}
	}
	
	option = {
		name = elven_celestial_grace_events.301.d
		ai_chance = {
			factor = 0
		}
		default_hide_option = yes
	}
}
