############################
# Main Story: Through the Shattered Glass #
############################

# Story: The Children of Anoriel are the remnants of a civilisation that successfully rebelled against their former god, Galtaereth. Upon severing their Covenant and seeing their patron weakened, their society split in two. Most were emboldened by their triumph and grew hungry for more - to commit opportunistic deicide and seize HER Divine Authority. Those who opposed that ambition were exiled, declared anathema, and forced to flee into the void of space. However, the opportunity turned out to be a cruel trick - a lure which the vengeful goddess used to lead HER treacherous disciples into a trap. It was only thanks to the intervention of another Shroud entity, Anoriel, that the exiles would elude annihilation. Though not even they were spared from the curse that rendered their souls mortal and consigned to eventually be consumed.
# Objectives: Withstand the curse that periodically culls the founder species' population. Grow strong enough to enter the former home cluster, and seal the Wound to cut off Galtaereth from this reality.

# Stage 0 (Gatekeeper Event)
# Start the event OR show an error message if cannot
country_event = {
	id = elven_anorim_origin_events.100
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_origin = origin_drow_prismatic_refuge
	}
	
	immediate = {
		if = {
			limit = {
				NOT = { is_same_value = event_target:anorim_star_owner_of_@event_target:anorim_star }
			}
			country_event = {
				id = elven_anorim_origin_events.0001
			}
		}
		else = {
			country_event = {
				id = elven_anorim_origin_events.101
			}
		}
	}
}

# Part 1 Stage 1 (Flickering Glimmer)
# Intro. Introduce and explain the situation.
country_event = {
	id = elven_anorim_origin_events.101
	title = "elven_anorim_origin_events.101.name"
	desc = "elven_anorim_origin_events.101.desc"
	picture = GFX_evt_cosmic_storms_particle_storm
	show_sound = event_mystic_reveal

	is_triggered_only = yes

	trigger = {
		has_origin = origin_drow_prismatic_refuge
	}
	
	immediate = {
		if = {
			limit = {
				exists = event_target:anorim_star
			}
			event_target:anorim_star = {
				# A very messy workaround to prevent dyson swarms from being built
				# Should be... relatively safe to use after galaxy generation
				set_star_flag = vela_system
			}
		}
		begin_event_chain = {
			event_chain = elven_anorim_prismatic_refuge_chain
			target = this
		}
		create_point_of_interest = {
			id = elven_anorim_prismatic_refuge_survive_poi
			name = "elven_anorim_prismatic_refuge_survive_poi"
			desc = elven_anorim_prismatic_refuge_survive_poi_desc
			event_chain = elven_anorim_prismatic_refuge_chain
			location = event_target:anorim_anoriel
		}
		if = {
			limit = {
				NOT = {
					any_situation = {
						is_situation_type = eos_anorim_situation_galtaereth
					}
				}
			}
			start_situation = {
				type = eos_anorim_situation_galtaereth
				target = event_target:anorim_anoriel
				effect = {
					save_event_target_as = eos_anorim_situation_galtaereth
					add_situation_progress = 600
				}
			}
		}
	}

	after = {
		hidden_effect = {
			create_variable_if_not_exists = {
				VARIABLE = eos_anorim_story
				VALUE_IF_CREATED = 0
			}
		}
	}
	
	option = {
		name = elven_anorim_origin_events.101.a
		trigger = {
			is_nomadic = yes
		}
	}
	option = {
		name = elven_anorim_origin_events.101.b
		trigger = {
			is_nomadic = no
		}
	}
}

# Part 2 Stage 1 (Creeping Gloom)
# Begin the second part of the event chain. Create the old home cluster and increase the intensity of Dark Tithe
country_event = {
	id = elven_anorim_origin_events.201
	title = "elven_anorim_origin_events.201.name"
	desc = "elven_anorim_origin_events.201.desc"
	picture = GFX_evt_cosmic_storms_nexus_storm
	show_sound = event_mystic_reveal

	is_triggered_only = yes

	trigger = {
		is_variable_set = eos_anorim_story
	}
	
	immediate = {
		if = {
			limit = {
				# Only one Cursed Cluster per galaxy!	
				NOT = { exists = event_target:anorim_evergloom_cluster_01 }
			}

			# Spawn the Evergloom systems
			# Their layout needs to be precise, so we're placing it as far away from the galactic rim as the game allows without breaking things
			set_spawn_system_batch = begin

			no_scope = {
				# Makes system positions originate from galactic core
				# We want it to be pretty far out so it doesn't intersect with The Chosen, etc.
				spawn_system = {
					min_distance >= 600
					max_distance <= 600
					min_orientation_angle = 135
					max_orientation_angle = 135
					initializer = eilistraee_event1_01_initializer
					hyperlane = no
					effect = {
						save_global_event_target_as = anorim_evergloom_cluster_01
						add_modifier = { modifier = eos_nebula_evergloom }
						# Hacky workarounds to prevent exploits
						set_star_flag = lcluster
						set_star_flag = vela_system
					}
				}
			}
			event_target:anorim_evergloom_cluster_01 = {
				spawn_system = {
					min_distance >= 20
					max_distance <= 30
					min_orientation_angle = 120
					max_orientation_angle = 120
					initializer = eilistraee_event1_02_initializer
					hyperlane = no
					effect = {
						save_global_event_target_as = anorim_evergloom_cluster_02
						add_modifier = { modifier = eos_nebula_evergloom }
						# Hacky workarounds to prevent exploits
						set_star_flag = lcluster
						set_star_flag = vela_system
					}
				}
			}
			event_target:anorim_evergloom_cluster_02 = {
				spawn_system = {
					min_distance >= 18
					max_distance <= 26
					min_orientation_angle = 60
					max_orientation_angle = 60
					initializer = eilistraee_event1_03_initializer
					hyperlane = no
					effect = {
						save_global_event_target_as = anorim_evergloom_cluster_03
						add_modifier = { modifier = eos_nebula_evergloom }
						# Hacky workarounds to prevent exploits
						set_star_flag = lcluster
						set_star_flag = vela_system
					}
				}
			}
			event_target:anorim_evergloom_cluster_03 = {
				spawn_system = {
					min_distance >= 15
					max_distance <= 25
					min_orientation_angle = 40
					max_orientation_angle = 40
					initializer = eilistraee_event1_04_initializer
					hyperlane = no
					effect = {
						save_global_event_target_as = anorim_evergloom_cluster_04
						add_modifier = { modifier = eos_nebula_evergloom }
						# Hacky workarounds to prevent exploits
						set_star_flag = lcluster
						set_star_flag = vela_system
					}
				}
			}
			event_target:anorim_evergloom_cluster_04 = {
				spawn_system = {
					min_distance >= 60
					max_distance <= 64
					min_orientation_angle = 15
					max_orientation_angle = 15
					initializer = eilistraee_event1_05_initializer
					hyperlane = no
					effect = {
						save_global_event_target_as = anorim_evergloom_cluster_05
						add_modifier = { modifier = eos_nebula_evergloom }
						# Hacky workarounds to prevent exploits
						set_star_flag = lcluster
						set_star_flag = vela_system
					}
				}
			}

			event_target:anorim_evergloom_cluster_03 = {
				create_nebula = {
					name = eos_nebula_evergloom
					radius = 62
					effect = {
						create_elven_nebula_red_purple = yes
						nebula_cloaking_effect = yes
					}
				}
			}

			event_target:anorim_evergloom_cluster_05 = {
				create_nebula = {
					name = eos_nebula_evergloom
					radius = 10
					effect = {
						create_elven_nebula_red_purple = yes
						nebula_cloaking_effect = yes
					}
				}
			}

			set_spawn_system_batch = end
		}
	}

	after = {
		hidden_effect = {
			set_variable = {
				which = eos_anorim_story
				value = 4
			}
		}
		country_event = {
			id = elven_anorim_origin_events.202
		}
	}
	
	option = {
		name = elven_anorim_origin_events.201.a
	}
}

	location = target

	}

		set_situation_progress = 0

		every_country = {

						has_trait = trait_drow_anorim_curse
					}
				}
				if = {
					limit = {
						solar_system = {
							is_same_value = event_target:anorim_star
						}
					}
					create_message = {
						type = EOS_ANORIM_GALTAERETH_EVADED
						localization = EOS_ANORIM_GALTAERETH_EVADED_DESC
						days = 30
						target = this.solar_system
						variable = {
							type = name
							localization = PLANET
							scope = this
						}
					}
				}
				else_if = {
					limit = {
						solar_system = {
							has_star_flag = eos_evergloom_cluster
						}
					}
					eos_evergloom_tithe = {
						MESSAGE = EOS_ANORIM_GALTAERETH_FOUND
						MESSAGE_DESC = EOS_ANORIM_GALTAERETH_FOUND_DESC
						PERCENTAGE = 0.20
					}
				}
				else = {
					eos_evergloom_tithe = {
						MESSAGE = EOS_ANORIM_GALTAERETH_FOUND
						MESSAGE_DESC = EOS_ANORIM_GALTAERETH_FOUND_DESC
						PERCENTAGE = 0.10
					}
				}
			}
		}
	}

	after = {
		hidden_effect = {
			# The first time it happens, progress the main story
			owner = {
				if = {
					limit = {
						is_variable_set = eos_anorim_story

							which = eos_anorim_story
							value < 1
						}
					}
					set_variable = {
						which = eos_anorim_story
						value = 1
					}
					country_event = {
						id = elven_anorim_origin_events.102
						days = 7
					}
				}
			}
		}
	}
