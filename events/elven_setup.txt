country_event = {
	id = elven_setup.41
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_species_class = ELVEN
		OR = {
			has_ethic = ethic_authoritarian
			has_ethic = ethic_xenophobe
		}
	}
	
	immediate = {
		set_policy = {
			policy = slavery
			option = slavery_not_allowed
			cooldown = no
		}
		set_policy = {
			policy = purge
			option = purge_displacement_only
			cooldown = no
		}
		set_policy = {
			policy = first_contact_protocol
			option = first_contact_cautious
			cooldown = no
		}
	}
}

# Disable caste system for authoritarian drow by default
event = {
	id = elven_setup.42
	hide_window = yes
	is_triggered_only = yes
	
	immediate = {
		every_country = {
			limit = {
				is_species_class = DROW
				has_ethic = ethic_authoritarian
			}

			save_event_target_as = drow_country

			owner_species = {
				set_citizenship_type = {
					country = event_target:drow_country
					type = citizenship_full
					cooldown = no
				}
			}
		}
	}
}
