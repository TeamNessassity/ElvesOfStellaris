# I don't want AI renaming my planets after they colonize.
planet_event = {
	id = elven_colony_events.11
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				has_owner = yes
				owner = {
					is_ai = yes
				}
				has_planet_flag = planet_polydora
			}
			
			set_name = "Halidrel'dár"
		}
		else_if = {
			limit = {
				has_owner = yes
				owner = {
					is_ai = yes
				}
				has_planet_flag = planet_narenwyn
			}
			
			set_name = "Narenwýn"
		}
		else_if = {
			limit = {
				has_owner = yes
				owner = {
					is_ai = yes
				}
				has_planet_flag = planet_morelyn
			}
			
			set_name = "Morelýn"
		}
	}
}

# Arcology project switches various unique deposits on elven planets
planet_event = {
	id = elven_colony_events.31

	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_planet_class = pc_city
		OR = {
			has_planet_flag = lunari_planet_flag
			has_planet_flag = eilistraeean_capital
			has_planet_flag = lhuren_capital
			has_planet_flag = ayleid_capital
		}
	}

	immediate = {
		# Delayed event that changes planet entity.
		# Arcology project immediately overwrites anything we do here, so we have to delay.
		planet_event = { id = elven_colony_events.32 days = 1 }

		if = {
			limit = {
				has_planet_flag = lunari_capital
				NOT = { has_planet_flag = removed_ice_blossoms }
			}
	
			add_deposit = d_lunari_planetary_cp_ice_blossoms
		}

		if = {
			limit = {
				has_planet_flag = eilistraeean_capital
			}
	
			add_deposit = d_drow_planetary_cheel_krezust

			if = {
				limit = { NOT = { has_planet_flag = removed_feral_fairies } }

				add_deposit = d_drow_planetary_cp_klepto_fairies
			}
		}
		
		if = {
			limit = {
				has_planet_flag = lhuren_capital
			}
	
			add_deposit = d_elven_planetary_forbidden_arcology

			if = {
				limit = { NOT = { has_planet_flag = removed_sentient_dragons } }

				add_deposit = d_elven_planetary_sentient_dragons
			}

			if = {
				limit = { NOT = { has_planet_flag = removed_gardens_of_heaven } }

				add_deposit = d_elven_planetary_gardens_of_heaven
			}
		}

		if = {
			limit = {
				has_planet_flag = ayleid_capital
			}
	
			add_deposit = d_elven_planetary_frensca_tol

			if = {
				limit = { NOT = { has_planet_flag = removed_tir_aranad } }

				add_deposit = d_elven_planetary_tir_aranad
			}
		}
	}
}

		name = elven_colony_events.41.a
	}
}

# Terraformed Permafrost worlds are reverted back to Permafrost after some time
planet_event = {
	id = elven_colony_events.42
	title = "elven_colony_events.42.name"
	desc = "elven_colony_events.42.desc"
	picture = GFX_evt_cold_barren
	show_sound = event_air_raid_siren
	location = ROOT

	is_triggered_only = yes

	trigger = {
		has_planet_flag = lunari_planet_flag
		NOR = {
			is_planet_class = pc_city
			is_planet_class = pc_lunari_city
			is_planet_class = pc_lunari_arctic
		}
    }

	immediate = {
		revert_planet_to_permafrost = yes

		# Nasty modifier as a bonus.
		# What, you thought you could break origin rules and get a Gaia world?
		add_modifier = {
			modifier = "lunari_ecological_retaliation"
			days = 7200
		}
	}

	option = {
		name = elven_colony_events.42.a
	}
}
